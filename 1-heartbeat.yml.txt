name: heartbeat
on:
  schedule:
    - cron: "23 */6 * * *"
  workflow_dispatch:
jobs:
  beat:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install pynostr websocket-client
      - run: python scripts/heartbeat.py
        env:
          NSEC: ${{ secrets.NSEC }}
